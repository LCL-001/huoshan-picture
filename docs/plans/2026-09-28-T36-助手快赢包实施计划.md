```sql
-- AI 助手会话索引（docs/plan.md T36，设计见 docs/designs/2026-09-28-助手快赢包设计.md）
-- 只记"谁发起过哪些对话"：标题取首条消息截断；消息体本身仍在引擎 chat_message，本表不冗余
use yu_picture;

SET NAMES utf8mb4;

create table if not exists ai_conversation
(
    id         bigint auto_increment comment 'id' primary key,
    userId     bigint                             not null comment '图库用户 id',
    chatId     varchar(64)                        not null comment '前端对话串标识（crypto.randomUUID，36 字符）',
    title      varchar(256)                       not null comment '会话标题（首条消息截断 64 字，后续不覆盖）',
    createTime datetime default CURRENT_TIMESTAMP not null comment '创建时间',
    updateTime datetime default CURRENT_TIMESTAMP not null on update CURRENT_TIMESTAMP comment '更新时间',
    isDelete   tinyint  default 0                 not null comment '是否删除',
    UNIQUE KEY uk_user_chat (userId, chatId),
    INDEX idx_user_update (userId, updateTime)
) comment 'AI 助手会话索引' collate = utf8mb4_unicode_ci;
```

```java
package com.lcl.yunpicturebackend.service.impl;

import cn.hutool.core.util.StrUtil;
import com.baomidou.mybatisplus.core.conditions.query.QueryWrapper;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.lcl.yunpicturebackend.domain.po.AiConversation;
import com.lcl.yunpicturebackend.domain.vo.AiConversationVO;
import com.lcl.yunpicturebackend.exception.ErrorCode;
import com.lcl.yunpicturebackend.exception.ThrowUtils;
import com.lcl.yunpicturebackend.mapper.AiConversationMapper;
import com.lcl.yunpicturebackend.service.IAiConversationService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.stereotype.Service;

import java.util.Date;
import java.util.List;
import java.util.stream.Collectors;

@Slf4j
@Service
public class AiConversationServiceImpl extends ServiceImpl<AiConversationMapper, AiConversation>
        implements IAiConversationService {

    private static final int TITLE_MAX_CHARS = 64;
    private static final long MAX_PAGE_SIZE = 50;

    @Override
    public void recordChat(Long userId, String chatId, String firstMessage) {
        if (userId == null || StrUtil.isBlank(chatId)) {
            // chatId 为空 = 引擎自起会话（前端总自带 chatId，这条只兜底），不进索引
            return;
        }
        AiConversation existing = baseMapper.selectOne(new QueryWrapper<AiConversation>()
                .eq("userId", userId).eq("chatId", chatId.trim()));
        Date now = new Date();
        if (existing != null) {
            AiConversation bump = new AiConversation();
            bump.setId(existing.getId());
            bump.setUpdateTime(now); // 显式刷时间，不依赖 on update CURRENT_TIMESTAMP 的空更新语义
            baseMapper.updateById(bump);
            return;
        }
        AiConversation row = new AiConversation();
        row.setUserId(userId);
        row.setChatId(chatId.trim());
        row.setTitle(truncate(firstMessage));
        row.setCreateTime(now);
        row.setUpdateTime(now);
        try {
            baseMapper.insert(row);
        } catch (DuplicateKeyException e) {
            // 并发首聊同一 chatId：唯一键兜底即"已存在"，本次不刷时间也无碍
            log.debug("ai_conversation 并发插入冲突（同一 chatId 已由其它请求建立）");
        }
    }

    @Override
    public List<AiConversationVO> listByUser(Long userId, long page, long pageSize) {
        long size = Math.min(Math.max(pageSize, 1), MAX_PAGE_SIZE);
        long offset = (Math.max(page, 1) - 1) * size;
        List<AiConversation> rows = baseMapper.selectList(new QueryWrapper<AiConversation>()
                .eq("userId", userId)
                .orderByDesc("updateTime")
                // 整数拼接无注入面；不走分页插件，避免依赖 MP 拦截器配置
                .last("LIMIT " + size + " OFFSET " + offset));
        return rows.stream().map(row -> AiConversationVO.builder()
                        .chatId(row.getChatId())
                        .title(row.getTitle())
                        .createTime(row.getCreateTime())
                        .updateTime(row.getUpdateTime())
                        .build())
                .collect(Collectors.toList());
    }

    @Override
    public void requireOwned(Long userId, String chatId) {
        ThrowUtils.throwIf(StrUtil.isBlank(chatId), ErrorCode.PARAMS_ERROR, "chatId 不能为空");
        Long count = baseMapper.selectCount(new QueryWrapper<AiConversation>()
                .eq("userId", userId).eq("chatId", chatId));
        ThrowUtils.throwIf(count == null || count == 0, ErrorCode.NOT_FOUND_ERROR, "会话不存在或无权访问");
    }

    private String truncate(String message) {
        String trimmed = StrUtil.trimToEmpty(message);
        if (trimmed.isEmpty()) {
            return "新对话";
        }
        return trimmed.length() <= TITLE_MAX_CHARS
                ? trimmed : trimmed.substring(0, TITLE_MAX_CHARS) + "…";
    }
}
```

```java
/**
 * 会话历史消息（T37）：归属判在本服务（索引表 user_id），引擎按 userId+chatId 派生确定性会话 id。
```

```java
/** 引擎读历史端点（T37）：与 /chat 同一内部密钥信道；读历史不需要用户凭据（引擎不碰图库） */
private static final String ENGINE_MESSAGES_PATH = "/ai/huoshan/conversations";

/**
 * 按 chatId 拉一场会话的历史消息（T37）。返回引擎响应体原文（BaseResponse JSON），
 * 由前端按 code 判读——代理不重新包装，避免 data 里再嵌一层转义 JSON。
 */
public String fetchConversationMessages(Long userId, String chatId) {
    String baseUrl = requireConfigured(properties.getEngineBaseUrl(), "图库助手引擎地址未配置");
    String apiKey = requireConfigured(properties.getInternalApiKey(), "图库助手引擎密钥未配置");
    HttpURLConnection connection = null;
    try {
        connection = (HttpURLConnection) new URL(messagesUrl(baseUrl, userId, chatId)).openConnection();
        connection.setRequestMethod("GET");
        // 与 /chat 同口径：绝不跟随重定向，避免内部密钥外泄
        connection.setInstanceFollowRedirects(false);
        connection.setRequestProperty(INTERNAL_API_KEY_HEADER, apiKey);
        connection.setConnectTimeout(toMillis(properties.getConnectTimeout()));
        connection.setReadTimeout(toMillis(properties.getReadTimeout()));
        int status = connection.getResponseCode();
        if (status != HttpURLConnection.HTTP_OK) {
            log.warn("AI 助手读历史响应异常, status={}", status);
            throw new BusinessException(ErrorCode.SYSTEM_ERROR, "引擎暂时不可用，请稍后重试");
        }
        return readAll(connection.getInputStream());
    } catch (IOException e) {
        log.warn("AI 助手读历史失败: {}", e.getMessage());
        throw new BusinessException(ErrorCode.SYSTEM_ERROR, "引擎暂时不可用，请稍后重试");
    } finally {
        if (connection != null) {
            connection.disconnect();
        }
    }
}

private String messagesUrl(String baseUrl, Long userId, String chatId) {
    return StrUtil.removeSuffix(baseUrl, "/") + ENGINE_MESSAGES_PATH
            + "/" + encode(chatId) + "/messages?userId=" + userId;
}

private String readAll(java.io.InputStream body) throws IOException {
    try (BufferedReader reader = new BufferedReader(new InputStreamReader(body, StandardCharsets.UTF_8))) {
        StringBuilder sb = new StringBuilder();
        String line;
        while ((line = reader.readLine()) != null) {
            if (sb.length() > 0) {
                sb.append('\n');
            }
            sb.append(line);
        }
        return sb.toString();
    }
}
```

```ts
/** 引擎历史消息（GET conversations/{chatId}/messages 的 data 元素）里前端用到的字段 */
export interface HistoryMessage {
  id: string
  role: string
  content: string
  steps?: { name?: string; content: string }[] | null
}

/**
 * 把引擎组装好的"轮"结构翻成页面 turns。ChatHistoryAssembler 保证 user/assistant 交替，
 * 防御两处：开头就是 assistant（没有归属提问）跳过；连续 assistant 并入最后一轮。
 * 历史步骤固定按工具步骤渲染（引擎只入库工具决策与工具响应，没有"思考"独立行）。
 */
export const assembleHistoryTurns = (messages: HistoryMessage[]): AssistantTurn[] => {
  const turns: AssistantTurn[] = []
  for (const message of messages) {
    if (message.role === 'user') {
      turns.push({ question: message.content, answer: '', steps: [], running: false })
      continue
    }
    if (message.role !== 'assistant' || turns.length === 0) {
      continue
    }
    const turn = turns[turns.length - 1]
    turn.answer = message.content
    turn.steps = (message.steps ?? []).map((step) => ({
      kind: 'tool' as const,
      name: step.name,
      content: step.content,
    }))
  }
  return turns
}
```

先跑该测试类确认红（实现仍发 answer），然后 `AiAssistantProxyManager.sendErrorAsAnswer` 改名并改帧：

```java
/**
 * 把失败合成一条 {@code error} 事件 + {@code [DONE]} 后收尾（T39）。
 * error 是 F13 协议的既有事件类型（引擎侧失败早已用它），代理侧不再伪装成模型说的话——
 * 前端据此渲染独立错误样式；这里不引入新事件类型、不给引擎加契约。
 */
private void sendErrorAsError(SseEmitter emitter, String text) {
    try {
        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("event", "error");
        payload.put("content", text);
        emitter.send(objectMapper.writeValueAsString(payload), MediaType.TEXT_PLAIN);
        emitter.send(DONE_FLAG, MediaType.TEXT_PLAIN);
    } catch (IOException | IllegalStateException e) {
        // 客户端已断开时收尾必然失败，属正常路径（连接已废弃，无处上报）
        log.debug("AI 助手事件收尾失败，客户端可能已断开");
    } finally {
        emitter.complete();
    }
}
```

`forward()` 里三处调用点同步改名。类头 javadoc 若有"协议仍只有 step/answer"的字样一并订正。
